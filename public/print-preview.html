// ==================== Edit Mode Logic ==================== let isEditMode = false; let currentPONumber = ''; let currentCategory = ''; // Initialize edit controls function initEditControls() { const toggleEditBtn = document.getElementById('toggleEditBtn'); const saveBtn = document.getElementById('saveBtn'); if (!toggleEditBtn || !saveBtn) return; toggleEditBtn.addEventListener('click', toggleEditMode); saveBtn.addEventListener('click', saveEdits); } function toggleEditMode() { isEditMode = !isEditMode; const container = document.getElementById('printOutput'); const toggleBtn = document.getElementById('toggleEditBtn'); const saveBtn = document.getElementById('saveBtn'); if (isEditMode) { container.classList.add('edit-mode'); toggleBtn.textContent = '取消编辑'; saveBtn.style.display = 'inline-block'; enableEditing(); } else { container.classList.remove('edit-mode'); toggleBtn.textContent = '编辑'; saveBtn.style.display = 'none'; disableEditing(); } } function enableEditing() { document.querySelectorAll('[contenteditable]').forEach(el => { el.setAttribute('contenteditable', 'true'); }); } function disableEditing() { document.querySelectorAll('[contenteditable]').forEach(el => { el.setAttribute('contenteditable', 'false'); }); } function saveEdits() { try { const editedData = collectEditedData(); saveToPO(currentPONumber, currentCategory, editedData); alert('✅ 保存成功！'); // Notify parent window to refresh if (window.opener && !window.opener.closed) { try { window.opener.postMessage({ type: 'PO_UPDATED', poNumber: currentPONumber }, '*'); } catch (e) { console.log('无法通知父窗口:', e); } } // Exit edit mode toggleEditMode(); } catch (error) { console.error('保存失败:', error); alert('❌ 保存失败: ' + error.message); } } function collectEditedData() { const rows = document.querySelectorAll('.print-table tbody tr:not(.total-row)'); const items = []; rows.forEach(row => { const itemIndex = row.dataset.itemIndex; if (!itemIndex) return; // Skip rows without index const item = { _index: parseInt(itemIndex) }; row.querySelectorAll('[data-field]').forEach(cell => { const field = cell.dataset.field; let value = cell.textContent.trim(); // Convert numbers if (field === 'qtyLeft' || field === 'qtyRight') { value = parseInt(value) || 0; } item[field] = value; }); items.push(item); }); return items; } function saveToPO(poNumber, category, editedData) { const allPOs = JSON.parse(localStorage.getItem('purchase_orders') || '[]'); const poIndex = allPOs.findIndex(p => p.poNumber === poNumber); if (poIndex === -1) { throw new Error('找不到采购单: ' + poNumber); } // Update items with edited data editedData.forEach(editedItem => { const originalIndex = editedItem._index; if (allPOs[poIndex].items[originalIndex]) { const original = allPOs[poIndex].items[originalIndex]; // Update fields if (editedItem.productModelName) original.productModelName = editedItem.productModelName; if (editedItem.spec) original.spec = editedItem.spec; if (editedItem.mb) original.mb = editedItem.mb; if (editedItem.remark !== undefined) original.remark = editedItem.remark; // Update quantity (combine left/right back to qty format) if (editedItem.qtyLeft !== undefined || editedItem.qtyRight !== undefined) { const left = editedItem.qtyLeft || 0; const right = editedItem.qtyRight || 0; original.qty = `${left}/${right}`; } } }); // Update metadata allPOs[poIndex].lastModified = new Date().toISOString(); localStorage.setItem('purchase_orders', JSON.stringify(allPOs)); console.log('✅ PO updated:', poNumber); } // Listen for messages from parent window window.addEventListener('message', (event) => { if (event.data.type === 'REFRESH_PO_LIST') { console.log('收到父窗口刷新请求'); } });
